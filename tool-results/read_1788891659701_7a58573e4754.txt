    48→type Dict = Record<string, unknown>;
    49→const isDict = (v: unknown): v is Dict => typeof v === 'object' && v !== null && !Array.isArray(v);
    50→
    51→function* walkDicts(o: unknown): Generator<Dict> {
    52→  if (isDict(o)) {
    53→    yield o;
    54→    for (const v of Object.values(o)) yield* walkDicts(v);
    55→  } else if (Array.isArray(o)) {
    56→    for (const v of o) yield* walkDicts(v);
    57→  }
    58→}
    59→
    60→/** Collect every value stored under `key` anywhere in the tree (e.g. all renderers of one kind). */
    61→function collect<T = Dict>(o: unknown, key: string): T[] {
    62→  const out: T[] = [];
    63→  for (const d of walkDicts(o)) {
    64→    const v = d[key];
    65→    if (v !== undefined) out.push(v as T);
    66→  }
    67→  return out;
    68→}

    70→function findFirst(o: unknown, pred: (d: Dict) => string | undefined): string | undefined {
    71→  for (const d of walkDicts(o)) {
    72→    const r = pred(d);
    73→    if (r) return r;
    74→  }
    75→  return undefined;
    76→}

    97→function pickThumb(o: unknown): string | undefined {
    98→  let best: { url: string; area: number } | undefined;
    99→  for (const d of walkDicts(o)) {
   100→    const th = d.thumbnails;
   101→    if (Array.isArray(th)) {
   102→      for (const t of th) {
   103→        if (isDict(t) && typeof t.url === 'string' && typeof t.width === 'number' && typeof t.height === 'number') {
   104→          const area = t.width * t.height;
   105→          if (!best || area > best.area) best = { url: t.url, area };
   106→        }
   107→      }
   108→    }
   109→  }
   110→  return best?.url;
   111→}

   115→/** "1.8B plays" / "9.1B views" / "31.1M subscribers" → number */
   116→export function parseCount(text: unknown): number {
   117→  if (typeof text !== 'string') return 0;
   118→  const m = /([\d.]+)\s*([KMB])?/i.exec(text.replace(/,/g, ''));
   119→  if (!m) return 0;
   120→  const n = parseFloat(m[1]);
   121→  if (Number.isNaN(n)) return 0;
   122→  const mult = m[2] ? { K: 1e3, M: 1e6, B: 1e9 }[m[2].toUpperCase()] ?? 1 : 1;
   123→  return Math.round(n * mult);
   124→}

   126→/** "3:41" → 221 (seconds) */
   127→export function parseDurationSec(text: unknown): number {
   128→  if (typeof text !== 'string') return 0;
   129→  const parts = text.trim().split(':').map((p) => parseInt(p, 10));
   130→  if (!parts.length || parts.some((p) => Number.isNaN(p))) return 0;
   131→  return parts.reduce((acc, p) => acc * 60 + p, 0);
   132→}
   133→
   134→const TIME_RUN_RE = /^\d{1,2}:[0-5]\d(?::[0-5]\d)?$/;
   135→
   136→/**
   137→ * Search/artist rows carry the track duration inside the subtitle runs
   138→ * (e.g. "Daft Punk • One More Time • 5:21"). Extracts the last m:ss run.
   139→ */
   140→function durationFromRuns(runs: unknown[]): number {
   141→  let last: string | undefined;
   142→  for (const r of runs) {
   143→    if (isDict(r) && typeof r.text === 'string' && TIME_RUN_RE.test(r.text.trim())) last = r.text.trim();
   144→  }
   145→  return last ? parseDurationSec(last) : 0;
   146→}

   166→/* ------------------------------ innertube ------------------------------- */
   167→
   168→async function innertube<T = Dict>(endpoint: string, body: Dict, ttlMs = 5 * 60_000): Promise<T> {
   169→  const cacheKey = `${endpoint}:${JSON.stringify(body)}`;
   170→  const hit = jsonCache.get(cacheKey);
   171→  if (hit && hit.exp > Date.now()) return hit.data as T;
   172→
   173→  const ctrl = new AbortController();
   174→  const timer = setTimeout(() => ctrl.abort(), 9000);
   175→  try {
   176→    const res = await fetch(`${INNERTUBE_BASE}/${endpoint}?key=${INNERTUBE_KEY}&prettyPrint=false`, {
   177→      method: 'POST',
   178→      headers: { 'Content-Type': 'application/json', 'User-Agent': USER_AGENT },
   179→      body: JSON.stringify({ ...CONTEXT, ...body }),
   180→      signal: ctrl.signal,
   181→      cache: 'no-store',
   182→    });
   183→    if (!res.ok) throw new Error(`innertube ${endpoint} → HTTP ${res.status}`);
   184→    const data = (await res.json()) as T;
   185→    if (jsonCache.size > 200) jsonCache.clear();
   186→    jsonCache.set(cacheKey, { exp: Date.now() + ttlMs, data });
   187→    return data;
   188→  } finally {
   189→    clearTimeout(timer);
   190→  }
   191→}

   202→export async function ytImage(url: string): Promise<{ buf: Uint8Array; type: string } | null> {
   203→  const hit = imageCache.get(url);
   204→  if (hit) return hit;
   205→  try {
   206→    const parsed = new URL(url);
   207→    if (parsed.protocol !== 'https:' || !IMG_HOST_ALLOW.some((re) => re.test(parsed.hostname))) return null;
   208→    const ctrl = new AbortController();
   209→    const timer = setTimeout(() => ctrl.abort(), 6000);
   210→    const res = await fetch(url, { signal: ctrl.signal, cache: 'no-store' });
   211→    clearTimeout(timer);
   212→    if (!res.ok) return null;
   213→    const type = res.headers.get('content-type') ?? 'image/jpeg';
   214→    if (!type.startsWith('image/')) return null;
   215→    const buf = new Uint8Array(await res.arrayBuffer());
   216→    if (buf.byteLength > 3_000_000) return null;
   217→    if (imageCache.size > 400) imageCache.clear();
   218→    const entry = { buf, type };
   219→    imageCache.set(url, entry);
   220→    return entry;
   221→  } catch {
   222→    return null;
   223→  }
   224→}

   226→/** Vibrant accent pair from a YouTube thumbnail (Spotify-style hero gradient). */
   227→export async function ytAccent(url: string | undefined): Promise<{ accent: string; accentSoft: string }> {
   228→  if (!url) return DEFAULT_ACCENT;
   229→  const hit = accentCache.get(url);
   230→  if (hit) return hit;
   231→  const img = await ytImage(url);
   232→  let result = DEFAULT_ACCENT;
   233→  if (img) {
   234→    try {
   235→      // Downscale to a 10×10 grid and pick the "vibrant" quadrant:
   236→      // mean of the brightest 25% of pixels (avoids black bars / dark edges).
   237→      const { data, info } = await sharp(Buffer.from(img.buf))
   238→        .removeAlpha()
   239→        .resize(10, 10, { fit: 'fill' })
   240→        .raw()
   241→        .toBuffer({ resolveWithObject: true });
   242→      const px: number[][] = [];
   243→      for (let i = 0; i + 2 < data.length; i += info.channels) {
   244→        px.push([data[i], data[i + 1], data[i + 2]]);
   245→      }
   246→      if (px.length) {
   247→        const lum = (p: number[]) => 0.2126 * p[0] + 0.7152 * p[1] + 0.0722 * p[2];
   248→        const sorted = [...px].sort((a, b) => lum(a) - lum(b));
   249→        const vibrant = sorted.slice(Math.floor(sorted.length * 0.75));
   250→        const mean = (arr: number[][]) => arr.reduce<number[]>((acc, p) => acc.map((v, i) => v + p[i]), [0, 0, 0]).map((v) => v / arr.length);
   251→        let [r, g, b] = mean(vibrant);
   252→        // keep the accent lively but not neon
   253→        const l = 0.2126 * r + 0.7152 * g + 0.0722 * b;
   254→        if (l < 60) {
   255→          r = Math.min(255, r * 2.4);
   256→          g = Math.min(255, g * 2.4);
   257→          b = Math.min(255, b * 2.4);
   258→        } else if (l > 180) {
   259→          r *= 0.72;
   260→          g *= 0.72;
   261→          b *= 0.72;
   262→        }
   263→        const hex = (n: number) => Math.round(Math.max(0, Math.min(255, n))).toString(16).padStart(2, '0');
   264→        const soft = (n: number) => Math.round(Math.max(36, Math.min(255, n * 0.4)));
   265→        result = {
   266→          accent: `#${hex(r)}${hex(g)}${hex(b)}`,
   267→          accentSoft: `#${hex(soft(r))}${hex(soft(g))}${hex(soft(b))}`,
   268→        };
   269→      }
   270→    } catch {
   271→      result = DEFAULT_ACCENT;
   272→    }
   273→  }
   274→  if (accentCache.size > 400) accentCache.clear();
   275→  accentCache.set(url, result);
   276→  return result;
   277→}

   294→function classify(browseId?: string, videoId?: string): RawKind {
   295→  if (browseId?.startsWith('UC')) return 'artist';
   296→  if (browseId?.startsWith('MPREb') || browseId?.startsWith('MPSP')) return 'album';
   297→  if (browseId?.startsWith('VL') || browseId?.startsWith('PL') || browseId?.startsWith('RD')) return 'playlist';
   298→  if (browseId) return 'unknown';
   299→  if (videoId) return 'song';
   300→  return 'unknown';
   301→}

   303→/** First subtitle run often carries the entity kind: "Song"/"Album"/"Video"/… */
   304→const KIND_WORDS: Record<string, RawKind> = {
   305→  Song: 'song',
   306→  Video: 'video',
   307→  Album: 'album',
   308→  Single: 'album',
   309→  EP: 'album',
   310→  Artist: 'artist',
   311→  Playlist: 'playlist',
   312→};
   313→
   314→/**
   315→ * Robust list-item classification. Songs can carry a *channel* browseId from
   316→ * their artist subtitle link, so explicit signals come first:
   317→ * playlistItemData / watch nav → song; browse nav → by prefix; then deep walks.
   318→ */
   319→function classifyListItem(r: Dict, kindWord?: string): RawKind {
   320→  if (kindWord && KIND_WORDS[kindWord]) return KIND_WORDS[kindWord];
   321→  const pid = (r.playlistItemData as Dict | undefined)?.videoId;
   322→  if (typeof pid === 'string') return 'song';
   323→  const nav = r.navigationEndpoint as Dict | undefined;
   324→  const navWe = (nav?.watchEndpoint as Dict | undefined)?.videoId;
   325→  if (typeof navWe === 'string') return 'song';
   326→  const navBe = (nav?.browseEndpoint as Dict | undefined)?.browseId;
   327→  if (typeof navBe === 'string') return classify(navBe, undefined);
   328→  const vid = findVideoId(r);
   329→  if (vid) return 'song';
   330→  const be = findBrowseId(r);
   331→  return classify(be, undefined);
   332→}

   338→/** Unwrap + parse a shelf's `contents` (each item is `{musicResponsiveListItemRenderer:{…}}`). */
   339→function shelfListItems(shelf: Dict): RawItem[] {
   340→  const contents = Array.isArray(shelf.contents) ? (shelf.contents as unknown[]) : [];
   341→  const out: RawItem[] = [];
   342→  for (const c of contents) {
   343→    if (!isDict(c)) continue;
   344→    const lr = (c.musicResponsiveListItemRenderer as Dict | undefined) ?? c;
   345→    const it = fromListItem(lr);
   346→    if (it) out.push(it);
   347→  }
   348→  return out;
   349→}

   351→/** Collect a page's track items: playlist shelf first, then bare album-style list items. */
   352→function collectTrackItems(d: Dict): RawItem[] {
   353→  for (const shelf of collect<Dict>(d, 'musicPlaylistShelfRenderer')) {
   354→    const items = shelfListItems(shelf).filter((i) => !!i.videoId);
   355→    if (items.length) return items;
   356→  }
   357→  const out: RawItem[] = [];
   358→  for (const lr of collect<Dict>(d, 'musicResponsiveListItemRenderer')) {
   359→    const pid = (lr.playlistItemData as Dict | undefined)?.videoId;
   360→    if (typeof pid !== 'string') continue;
   361→    const it = fromListItem(lr);
   362→    if (it?.videoId) out.push(it);
   363→  }
   364→  return out;
   365→}

   378→/** Unified page header parser (immersive + responsive header flavors). */
   379→function parseHeader(d: Dict): ParsedHeader {
   380→  const h =
   381→    collect<Dict>(d, 'musicResponsiveHeaderRenderer')[0] ??
   382→    collect<Dict>(d, 'musicImmersiveHeaderRenderer')[0] ??
   383→    collect<Dict>(d, 'musicHeaderRenderer')[0] ??
   384→    collect<Dict>(d, 'playlistHeaderRenderer')[0] ??
   385→    {};
   386→  const subtitleRuns = Array.isArray((h.subtitle as Dict | undefined)?.runs)
   387→    ? ((h.subtitle as Dict).runs as unknown[])
   388→    : [];
   389→  const strapRuns = Array.isArray((h.straplineTextOne as Dict | undefined)?.runs)
   390→    ? ((h.straplineTextOne as Dict).runs as unknown[])
   391→    : [];
   392→  let description = runsText((h.description as Dict | undefined)?.runs);
   393→  if (!description) {
   394→    const nested = (h.description as Dict | undefined)?.musicDescriptionShelfRenderer as Dict | undefined;
   395→    if (isDict(nested)) description = runsText((nested.description as Dict | undefined)?.runs);
   396→  }
   397→  const mlc = h.monthlyListenerCount as Dict | undefined;
   398→  const listeners = mlc ? parseCount(runsText(mlc.runs)) : undefined;
   399→  return {
   400→    title: runsText((h.title as Dict | undefined)?.runs),
   401→    subtitleRuns,
   402→    strapRuns,
   403→    secondSubtitle: runsText((h.secondSubtitle as Dict | undefined)?.runs),
   404→    description,
   405→    listeners,
   406→    thumb: pickThumb(h) ?? pickThumb(d),
   407→  };
   408→}

   410→/** musicResponsiveListItemRenderer (search shelves, album/playlist track lists, top songs) */
   411→function fromListItem(r: Dict): RawItem | null {
   412→  const flex = Array.isArray(r.flexColumns) ? (r.flexColumns as unknown[]) : [];
   413→  const colRuns = (i: number): unknown[] => {
   414→    const c = flex[i] as Dict | undefined;
   415→    const fc = c?.musicResponsiveListItemFlexColumnRenderer as Dict | undefined;
   416→    const text = fc?.text as Dict | undefined;
   417→    return Array.isArray(text?.runs) ? (text.runs as unknown[]) : [];
   418→  };
   419→  const title = runsText(colRuns(0));
   420→  if (!title) return null;
   421→
   422→  const fixed = Array.isArray(r.fixedColumns) ? (r.fixedColumns[0] as Dict | undefined) : undefined;
   423→  const durRuns = ((fixed?.musicResponsiveListItemFixedColumnRenderer as Dict | undefined)?.text as Dict | undefined)?.runs;
   424→  const durationText = runsText(durRuns) || undefined;
   425→
   426→  const pid = (r.playlistItemData as Dict | undefined)?.videoId;
   427→  const videoId = typeof pid === 'string' ? pid : findVideoId(r);
   428→  const navBrowse = ((r.navigationEndpoint as Dict | undefined)?.browseEndpoint as Dict | undefined)?.browseId;
   429→  const browseId = typeof navBrowse === 'string' ? navBrowse : findBrowseId(r);
   430→
   431→  const playsText = runsText(colRuns(2)) || undefined;
   432→  const subRuns1 = colRuns(1);
   433→  const firstSubRun = subRuns1.length ? subRuns1[0] : undefined;
   434→  const subKind =
   435→    isDict(firstSubRun) && typeof firstSubRun.text === 'string' ? firstSubRun.text : undefined;
   436→  return {
   437→    kind: classifyListItem(r, subKind),
   438→    videoId,
   439→    browseId,
   440→    title,
   441→    subtitleRuns: colRuns(1),
   442→    playsText: /play|view/i.test(playsText ?? '') ? playsText : undefined,
   443→    durationText,
   444→    thumb: pickThumb(r),
   445→  };
   446→}

   448→/** musicTwoRowItemRenderer (home shelves, artist discography, moods) */
   449→function fromTwoRow(t: Dict): RawItem | null {
   450→  const title = runsText((t.title as Dict | undefined)?.runs);
   451→  if (!title) return null;
   452→  const nav = (t.navigationEndpoint ?? t.onTap) as Dict | undefined;
   453→  const be = (nav?.browseEndpoint as Dict | undefined)?.browseId;
   454→  const we = nav?.watchEndpoint as Dict | undefined;
   455→  const wpe = nav?.watchPlaylistEndpoint as Dict | undefined;
   456→  const browseId =
   457→    typeof be === 'string' ? be : typeof wpe?.playlistId === 'string' ? (wpe.playlistId as string) : findBrowseId(t.title);
   458→  const videoId = typeof we?.videoId === 'string' ? (we.videoId as string) : undefined;
   459→  const subRuns = (t.subtitle as Dict | undefined)?.runs;
   460→  return {
   461→    kind: classify(browseId, videoId),
   462→    videoId,
   463→    browseId,
   464→    title,
   465→    subtitleRuns: Array.isArray(subRuns) ? (subRuns as unknown[]) : [],
   466→    playsText: undefined,
   467→    durationText: undefined,
   468→    thumb: pickThumb(t),
   469→  };
   470→}

   472→/** musicCardShelfRenderer (search top result) */
   473→function fromCard(c: Dict): RawItem | null {
   474→  const title = runsText((c.title as Dict | undefined)?.runs);
   475→  if (!title) return null;
   476→  const nav = (c.onTap ?? c.navigationEndpoint) as Dict | undefined;
   477→  const be = (nav?.browseEndpoint as Dict | undefined)?.browseId;
   478→  const we = nav?.watchEndpoint as Dict | undefined;
   479→  const browseId = typeof be === 'string' ? be : findBrowseId(c);
   480→  const videoId = typeof we?.videoId === 'string' ? (we.videoId as string) : findVideoId(c);
   481→  const subRuns = (c.subtitle as Dict | undefined)?.runs;
   482→  // "Song • Artist" / "Video • Artist" cards are tracks even when the card
   483→  // also carries an artist browse id (e.g. auto-generated meme "artists").
   484→  const subText = runsText(subRuns);
   485→  const kindWord = (subText.split('•')[0] ?? '').trim().toLowerCase();
   486→  const kind: RawKind =
   487→    (kindWord === 'song' || kindWord === 'video') && videoId ? 'song' : classify(browseId, videoId);
   488→  return {
   489→    kind,
   490→    videoId,
   491→    browseId,
   492→    title,

   519→function artistRefFrom(runs: unknown[]): { id: string; slug: string; name: string } {
   520→  for (const r of runs) {

   759→/** Live YouTube Music search, mapped into our SearchDTO shape. */
   760→export async function ytSearch(query: string): Promise<SearchDTO> {
   761→  // Songs-filtered search in parallel: song rows there carry real durations
   762→  // ("Daft Punk • One More Time • 5:21") which the mixed search omits.
   763→  const songsFilterPromise = innertube('search', { query, params: SEARCH_SONGS_FILTER })
   764→    .then((d) => d as Dict | null)
   765→    .catch(() => null);
   766→  const d = await innertube('search', { query });
   767→
   768→  let topResult: SearchDTO['topResult'] = null;
   769→  const card = collect<Dict>(d, 'musicCardShelfRenderer')[0];
   770→  if (isDict(card)) {
   771→    const item = fromCard(card);
   772→    if (item) topResult = await mapTopResult(item);
   773→  }
   774→
   775→  /* Collect items from BOTH layouts:
   776→     - shelf layout (filtered searches): musicShelfRenderer with titled buckets
   777→     - flat layout (mixed search): one itemSectionRenderer per result        */
   778→  const raw: { item: RawItem; shelfTitle: string }[] = [];
   779→  for (const shelf of collect<Dict>(d, 'musicShelfRenderer')) {
   780→    const shelfTitle = runsText((shelf.title as Dict | undefined)?.runs).toLowerCase();
   781→    for (const it of shelfListItems(shelf)) raw.push({ item: it, shelfTitle });
   782→  }
   783→  for (const isr of collect<Dict>(d, 'itemSectionRenderer')) {
   784→    const contents = Array.isArray(isr.contents) ? (isr.contents as unknown[]) : [];
   785→    for (const c of contents) {
   786→      const lr = isDict(c) ? (c.musicResponsiveListItemRenderer as Dict | undefined) : undefined;
   787→      if (isDict(lr)) {
   788→        const it = fromListItem(lr);
   789→        if (it) raw.push({ item: it, shelfTitle: '' });
   790→      }
   791→    }
   792→  }
   793→
   794→  const songs: RawItem[] = [];
   795→  const videos: RawItem[] = [];
   796→  const artists: RawItem[] = [];
   797→  const albums: RawItem[] = [];
   798→  const playlists: RawItem[] = [];
   799→  for (const { item, shelfTitle } of raw) {
   800→    const asVideo = shelfTitle.includes('video') && (item.kind === 'song' || item.kind === 'video');
   801→    switch (item.kind) {
   802→      case 'song':
   803→        if (asVideo) videos.push(item);
   804→        else songs.push(item);
   805→        break;
   806→      case 'video':
   807→        videos.push(item);
   808→        break;
   809→      case 'artist':
   810→        artists.push(item);
   811→        break;
   812→      case 'album':
   813→        albums.push(item);
   814→        break;
   815→      case 'playlist':
   816→        /* Channel-backed "playlists" (UC ids) have no browsable track list. */
   817→        if (/^(VL|PL|RD)/.test(item.browseId ?? '')) playlists.push(item);
   818→        break;
   819→      default:
   820→        break;
   821→    }
   822→  }
   823→
   824→  // Songs first (canonical topic uploads), then video uploads — user/VEVO
   825→  // uploads of the same tracks often behave differently for embedding, so
   826→  // keeping both improves playback odds and alternate resolution. Podcast
   827→  // episodes are dropped.
   828→  const musicVideos = videos.filter((v) => !isEpisodeItem(v));
   829→  const trackItems = [...songs.slice(0, 9), ...musicVideos.slice(0, 6)];
   830→  const durationMap = new Map<string, number>();
   831→  const songsFilter = await songsFilterPromise;
   832→  if (songsFilter) {
   833→    for (const shelf of collect<Dict>(songsFilter, 'musicShelfRenderer')) {
   834→      for (const it of shelfListItems(shelf)) {
   835→        if (!it.videoId) continue;
   836→        const dur = parseDurationSec(it.durationText) || durationFromRuns(it.subtitleRuns);
   837→        if (dur > 0) durationMap.set(it.videoId, dur);
   838→      }
   839→    }
   840→  }
   841→  const tracks = (
   842→    await Promise.all(
   843→      trackItems.map((i) => ytTrackDTO(i, { durationOverride: i.videoId ? durationMap.get(i.videoId) : undefined }))
   844→    )
   845→  ).filter((t): t is TrackDTO => t !== null);
   846→  const albumDtos = (await Promise.all(albums.slice(0, 8).map(ytAlbumDTO))).filter((a): a is AlbumDTO => a !== null);
   847→  const artistDtos = artists
   848→    .slice(0, 8)
   849→    .map(ytArtistDTO)
   850→    .filter((a): a is ArtistDTO => a !== null);
   851→  const playlistDtos = playlists
   852→    .slice(0, 8)
   853→    .map(ytPlaylistDTO)
   854→    .filter((p): p is PlaylistDTO => p !== null);
   855→
   856→  if (!topResult && tracks.length) topResult = { ...tracks[0], type: 'track' };
   857→  if (!topResult && artistDtos.length) topResult = { ...artistDtos[0], type: 'artist' };
   858→  if (!topResult && playlistDtos.length) topResult = { ...playlistDtos[0], type: 'playlist' };
   859→  if (!topResult && albumDtos.length) topResult = { ...albumDtos[0], type: 'album' };
   860→
   861→  return { query, topResult, tracks, albums: albumDtos, artists: artistDtos, playlists: playlistDtos };
   862→}

   864→/** Live YouTube Music home shelves (new releases, trending…). */

   867→  const shelves: ShelfDTO[] = [];

   870→      (((c.header as Dict | undefined)?.musicCarouselShelfBasicHeaderRenderer as Dict | undefined)?.title as Dict | undefined)?.runs

   882→      }
   883→      const listItem = content.musicResponsiveListItemRenderer as Dict | undefined;
   884→      if (isDict(listItem)) {
   885→        const it = fromListItem(listItem);
   886→        if (it) rawItems.push(it);

   894→    const seen = new Set<string>();

   896→    if (unique.length) shelves.push({ id: `yt-home-${slugify(title)}`, title, items: unique });
   897→  }

  1056→/** Live playlist page (VL/PL/RD ids). */
  1057→export async function ytPlaylist(slug: string): Promise<PlaylistDetailDTO | null> {
  1058→  const browseId = slug.replace(/^yt[-:]/, '');
  1059→  if (browseId.startsWith('single-')) return ytSingle(browseId.replace('single-', ''));
  1060→  const d = await innertube('browse', { browseId }, 15 * 60_000);
  1061→  const header = parseHeader(d);
  1062→  const name = header.title || 'Playlist';
  1063→  const accent = await ytAccent(header.thumb);
  1064→
  1065→  const items = collectTrackItems(d).slice(0, 50);
  1066→  const tracks = (
  1067→    await Promise.all(items.map((i, idx) => ytTrackDTO(i, { number: idx + 1 })))
  1068→  ).filter((t): t is TrackDTO => t !== null);
  1069→
  1070→  const owner = displayOwner(runsText(header.strapRuns));
  1071→  return {
  1072→    id: `yt:${browseId}`,
  1073→    slug: `yt-${browseId}`,
  1074→    name,
  1075→    description: header.description || null,
  1076→    cover: proxiedImage(header.thumb) || tracks[0]?.album.cover || null,
  1077→    coverFrom: accent.accentSoft,
  1078→    coverTo: null,
  1079→    icon: null,
  1080→    owner,
  1081→    editable: false,
  1082→    trackCount: tracks.length,
  1083→    totalDuration: tracks.reduce((s, t) => s + t.duration, 0),
  1084→    tracks,
  1085→    source: 'youtube',
